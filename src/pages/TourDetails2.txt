function useOnceVisible(threshold = 0.08) {
  const ref = useRef(null);
  const [vis, setVis] = useState(false);
  useEffect(() => {
    const el = ref.current; if (!el) return;
    const obs = new IntersectionObserver(([e]) => {
      if (e.isIntersecting) { setVis(true); obs.disconnect(); }
    }, { threshold });
    obs.observe(el); return () => obs.disconnect();
  }, [threshold]);
  return [ref, vis];
}

function Reveal({ children, delay = 0, y = 32, className = "" }) {
  const [ref, vis] = useOnceVisible();
  return (
    <div ref={ref} className={className} style={{
      opacity: vis ? 1 : 0,
      transform: vis ? "translateY(0)" : `translateY(${y}px)`,
      transition: `opacity 0.75s ease ${delay}s, transform 0.75s cubic-bezier(0.16,1,0.3,1) ${delay}s`,
    }}>
      {children}
    </div>
  );
}

                    style={{ background: "linear-gradient(135deg,#052e16,#0f766e)" }}>
                    <div className="absolute inset-0 opacity-10"
                      style={{ backgroundImage: "radial-gradient(circle at 80% 20%,white 0%,transparent 50%)" }} />
                    <p className="text-emerald-300/60 text-[10px] font-bold tracking-[.3em] uppercase mb-1 relative z-10" style={{ fontFamily: sans }}>Starting from</p>
                    <p className="text-yellow-400 font-black relative z-10" style={{ fontFamily: serif, fontSize: "2.4rem", lineHeight: 1 }}>{firstPrice}</p>
                    <p className="text-emerald-200/40 text-xs mt-1 relative z-10" style={{ fontFamily: sans }}>per person · varies by group size</p>
                  </div>
                  <div className="p-4 bg-white">
                    <p className="text-slate-400 text-[10px] font-bold tracking-widest uppercase mb-3" style={{ fontFamily: sans }}>All options</p>
                    <div className="space-y-1.5">
                      {tour.pricing.map((row, i) => {
                        const isNote = !row.price && !row.range;
                        return isNote ? (
                          <p key={i} className="text-slate-300 text-[10px] text-center pt-2" style={{ fontFamily: sans }}>{row.pkg}</p>
                        ) : (
                          <div key={i} className="flex items-center justify-between gap-2 p-2.5 rounded-xl"
                            style={{ background: i === 0 ? "#f0fdf4" : "#fafafa", border: `1px solid ${i === 0 ? "#bbf7d0" : "#f1f5f9"}` }}>
                            <span className="text-slate-600 text-xs leading-snug flex-1" style={{ fontFamily: sans }}>{row.pax || row.pkg}</span>
                            <span className="font-black text-emerald-700 text-sm flex-shrink-0" style={{ fontFamily: serif }}>{row.price || row.range}</span>
                          </div>
                        );
                      })}
                    </div>
                  </div>
                  <div className="px-4 pb-5 bg-white">
                    <button onClick={() => wa(`Hello KiriTour! I'd like to book: ${tour.title} (${tour.duration})`)}
                      className="w-full flex items-center justify-center gap-2.5 py-4 rounded-2xl font-black text-emerald-900 text-base hover:scale-105 active:scale-95 transition-all"
                      style={{ background: "linear-gradient(135deg,#facc15,#fbbf24,#f59e0b)", boxShadow: "0 6px 28px rgba(250,204,21,.45)", fontFamily: sans }}>
                      <Ico n="wa" s={18} c="#14532d" /> Book via WhatsApp
                    </button>
                    <p className="text-center text-slate-400 text-[10px] mt-2" style={{ fontFamily: sans }}>We reply within 2h · Custom itineraries available</p>
                  </div>
                </div>
              </Reveal>

              {/* Share */}
              <Reveal delay={0.2}>
                <button onClick={handleShare}
                  className="w-full flex items-center gap-3 px-5 py-4 rounded-2xl transition-all hover:scale-[1.02] active:scale-95"
                  style={{ background: copied ? "#f0fdf4" : "white", border: `1.5px solid ${copied ? "#86efac" : "#e2e8f0"}`, boxShadow: "0 2px 12px rgba(0,0,0,.04)" }}>
                  <div className="w-9 h-9 rounded-xl flex items-center justify-center flex-shrink-0" style={{ background: copied ? "#dcfce7" : "#f8fafc" }}>
                    <Ico n="share" s={16} c={copied ? "#16a34a" : "#94a3b8"} />
                  </div>
                  <div className="text-left">
                    <p className="font-bold text-sm text-slate-700" style={{ fontFamily: sans }}>{copied ? "Link copied!" : "Share this tour"}</p>
                    <p className="text-slate-400 text-[10px]" style={{ fontFamily: sans }}>Unique UUID link · Share anywhere</p>
                  </div>
                </button>
              </Reveal>

              {/* Help */}
              <Reveal delay={0.22}>
                <div className="rounded-2xl p-4" style={{ background: "white", border: "1px solid #f1f5f9" }}>
                  <p className="text-slate-400 text-[10px] uppercase tracking-widest mb-3" style={{ fontFamily: sans }}>Need help choosing?</p>
                  <button onClick={() => wa("Hello! I need help choosing the right tour for me.")}
                    className="w-full flex items-center justify-center gap-2 py-3 rounded-xl font-semibold text-emerald-700 text-sm hover:bg-emerald-50 transition-all"
                    style={{ fontFamily: sans, border: "1.5px solid #bbf7d0" }}>
                    <Ico n="wa" s={16} c="#16a34a" /> Chat with our team
                  </button>
                </div>
              </Reveal>
            </div>
          </div>
        </div>
      </div>

      <style>{`
        @keyframes td-kb {
          from { transform: scale(1.07); }
          to   { transform: scale(1); }
        }
        @keyframes td-scroll {
          0%   { transform: translateY(-100%); }
          100% { transform: translateY(290%); }
        }
      `}</style>
    </div>
  );
